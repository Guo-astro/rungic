cmake_minimum_required(VERSION 3.22)
project(RungicAgentSession LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Core Network WebSockets)
find_package(PkgConfig REQUIRED)
pkg_check_modules(GST REQUIRED IMPORTED_TARGET gstreamer-1.0 gstreamer-app-1.0)
add_library(session-core core.cpp)
target_link_libraries(session-core PUBLIC Qt6::Core)
add_executable(rungic-agent-session main.cpp session.cpp audio.cpp)
target_link_libraries(rungic-agent-session PRIVATE session-core Qt6::Network Qt6::WebSockets PkgConfig::GST)
add_executable(rungic-task-tools task-tools.cpp)
target_link_libraries(rungic-task-tools PRIVATE Qt6::Core)
include(CTest)
if(BUILD_TESTING)
    add_executable(session-core-test core-test.cpp)
    target_link_libraries(session-core-test PRIVATE session-core)
    add_test(NAME session-core COMMAND session-core-test)
    add_executable(session-state-test session-test.cpp session.cpp audio.cpp)
    target_link_libraries(session-state-test PRIVATE session-core Qt6::Network Qt6::WebSockets PkgConfig::GST)
    add_test(NAME session-state COMMAND session-state-test)
    add_executable(session-cloud-probe cloud-probe.cpp session.cpp audio.cpp)
    target_link_libraries(session-cloud-probe PRIVATE session-core Qt6::Network Qt6::WebSockets PkgConfig::GST)
endif()
install(TARGETS rungic-agent-session rungic-task-tools RUNTIME DESTINATION bin)
