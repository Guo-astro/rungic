cmake_minimum_required(VERSION 3.22)
project(moto_plasma_rime LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Core Qml VirtualKeyboard)
find_package(PkgConfig REQUIRED)
pkg_check_modules(RIME REQUIRED IMPORTED_TARGET rime)
add_library(motorimeplugin MODULE plugin.cpp)
target_link_libraries(motorimeplugin PRIVATE Qt6::Qml Qt6::VirtualKeyboard PkgConfig::RIME)
install(TARGETS motorimeplugin LIBRARY DESTINATION lib/moto-rime/qml/Moto/Rime)
install(FILES qmldir DESTINATION lib/moto-rime/qml/Moto/Rime)
add_executable(moto-rime-check check.cpp)
target_link_libraries(moto-rime-check PRIVATE Qt6::Core PkgConfig::RIME)
install(TARGETS moto-rime-check RUNTIME DESTINATION libexec)
