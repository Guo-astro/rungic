cmake_minimum_required(VERSION 3.22)
project(rungic_plasma_rime LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Core Qml VirtualKeyboard)
find_package(PkgConfig REQUIRED)
pkg_check_modules(RIME REQUIRED IMPORTED_TARGET rime)
add_library(rungicrimeplugin MODULE plugin.cpp)
target_link_libraries(rungicrimeplugin PRIVATE Qt6::Qml Qt6::VirtualKeyboard PkgConfig::RIME)
install(TARGETS rungicrimeplugin LIBRARY DESTINATION lib/rungic-rime/qml/Rungic/Rime)
install(FILES qmldir DESTINATION lib/rungic-rime/qml/Rungic/Rime)
add_executable(rungic-rime-check check.cpp)
target_link_libraries(rungic-rime-check PRIVATE Qt6::Core PkgConfig::RIME)
install(TARGETS rungic-rime-check RUNTIME DESTINATION libexec)
